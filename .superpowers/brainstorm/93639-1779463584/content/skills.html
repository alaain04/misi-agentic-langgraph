Skill contract — what every skill must implement (Python)

from abc import ABC, abstractmethod from dataclasses import dataclass from typing import Sequence @dataclass class SkillContext: dep_name: str hypothesis_id: str hypothesis: str # the hypothesis statement sbom: dict repo_path: str | None concern: str services: dict # injected: container, dao, mcp_client… class InvestigationSkill(ABC): id: str name: str description: str trigger_conditions: list[str] required_inputs: list[str] evidence_kinds: list[str] @abstractmethod async def execute(self, ctx: SkillContext) -> Sequence[Evidence]: """Gather evidence. Never raises — returns empty list on failure.""" ... def can_run(self, ctx: SkillContext) -> bool: """Check required_inputs are available before dispatching.""" return all(getattr(ctx, f, None) is not None for f in self.required_inputs)

Concrete example — VulnerabilitySkill wrapping existing Trivy logic

class VulnerabilitySkill(InvestigationSkill): id = "VulnerabilitySkill" name = "Vulnerability Surface Assessment" description = "Scans for known CVEs and security advisories" trigger_conditions = ["security concern", "CVE", "supply chain"] required_inputs = ["repo_path"] evidence_kinds = ["vulnerability"] async def execute(self, ctx: SkillContext) -> list[Evidence]: # existing Trivy logic, now wrapped to return typed Evidence scan_data = await run_trivy(ctx.services["container"], ctx.repo_path) vulns = _extract_vulns(scan_data, ctx.dep_name) return [ Evidence( id = uuid4().hex, kind = "vulnerability", dep_name = ctx.dep_name, skill_id = self.id, hypothesis_id = ctx.hypothesis_id, signal = f"{v.cve_id} ({v.severity}) in {ctx.dep_name}@{v.version}", raw_data = v.__dict__, source = "trivy", reliability = 0.95, confidence = _severity_to_confidence(v.severity), severity = v.severity.lower(), supports_hypothesis = True, ) for v in vulns ]

Skill registry — how skills are registered and discovered

# src/main_graph/skills/registry.py SKILL_REGISTRY: dict[str, InvestigationSkill] = { skill.id: skill for skill in [ VulnerabilitySkill(), MaintainerTrustSkill(), SupplyChainSkill(), LicenseSkill(), ReachabilitySkill(), BlastRadiusSkill(), ReleaseAnomalySkill(), EcosystemSkill(), ] } # Planner receives this as context — it selects by id SKILL_DESCRIPTIONS = { sid: f"{s.name}: {s.description} | triggers: {', '.join(s.trigger_conditions)}" for sid, s in SKILL_REGISTRY.items() }

Migration insight: Every existing ingestion subgraph becomes a skill. The Trivy scan logic, GitHub MCP calls, impact analysis — all preserved. What changes is the wrapper: instead of returning a result_id saved to MongoDB, each skill returns Evidence[] that flows into the shared state. The DAO persistence layer can still be kept for archiving, but it's no longer the primary data path.
